<script>
(function(){
  var script=document.currentScript;
  var card=script&&script.previousElementSibling;
  while(card&&!(card.classList&&card.classList.contains('cosmos-world-bar'))){card=card.previousElementSibling;}
  if(!card){var cards=document.querySelectorAll('.cosmos-world-bar');card=cards[cards.length-1];}
  if(!card)return;

  // Theme switch, same behavior as the original script.
  var KEY='cosmos-cw-theme';
  function safeGet(){try{return localStorage.getItem(KEY)||'dark';}catch(e){return 'dark';}}
  function safeSet(v){try{localStorage.setItem(KEY,v);}catch(e){}}
  function applyTheme(t){document.querySelectorAll('.cosmos-world-bar').forEach(function(el){el.setAttribute('data-theme',t);});}
  applyTheme(safeGet());
  document.querySelectorAll('.cosmos-world-bar').forEach(function(item){
    var btn=item.querySelector('.cosmos-cw-theme-btn');
    if(!btn||btn.dataset.cwThemeBound)return;
    btn.dataset.cwThemeBound='1';
    btn.addEventListener('click',function(ev){
      ev.preventDefault();ev.stopPropagation();
      var cur=item.getAttribute('data-theme')||'dark';
      var nxt=cur==='dark'?'light':'dark';
      safeSet(nxt);applyTheme(nxt);
    });
  });

  // Keep the original single-window player model that was already playable.
  if(!window._cw){
    window._cw={audio:new Audio(),playId:0,timer:null,activeBtn:null};
    window._cw.audio.loop=true;
    window._cw.audio.addEventListener('pause',function(){
      if(window._cw.activeBtn)window._cw.activeBtn.textContent='\u25b6';
    });
    window._cw.audio.addEventListener('play',function(){
      if(window._cw.activeBtn)window._cw.activeBtn.textContent='\u23f8';
    });
  }

  var lyricEl=card.querySelector('.cosmos-music-text');
  var playBtn=card.querySelector('.cosmos-music-btn');
  if(!lyricEl||!playBtn)return;

  var cw=window._cw;
  var audio=cw.audio;
  var loaded=false;
  var instanceId='cw-'+Date.now()+'-'+Math.random().toString(36).slice(2);
  playBtn.dataset.cwMusicId=instanceId;

  var txt=lyricEl.textContent.trim();
  var m=txt.match(/[《<]([^》>]+)[》>]\s*(.+)?/);
  if(!m||!m[1]){playBtn.style.display='none';return;}
  var keyword=m[2]?m[1].trim()+' '+m[2].trim():m[1].trim();

  function stopLocal(){
    try{clearTimeout(cw.timer);}catch(e){}
    try{audio.pause();audio.src='';}catch(e){}
    if(cw.activeBtn)cw.activeBtn.textContent='\u25b6';
    if(cw.activeBtn)cw.activeBtn.disabled=false;
    cw.activeBtn=null;
  }
  function announceStop(){
    var payload={type:'stop',id:instanceId,time:Date.now()};
    try{document.dispatchEvent(new CustomEvent('cosmos-world-music-stop',{detail:payload}));}catch(e){}
    try{localStorage.setItem('cosmos-world-music-stop',JSON.stringify(payload));}catch(e){}
    try{
      if(!window._cwBroadcast)window._cwBroadcast=new BroadcastChannel('cosmos-world-music-control');
      window._cwBroadcast.postMessage(payload);
    }catch(e){}
  }
  function handleExternalStop(payload){
    if(!payload||payload.id===instanceId)return;
    stopLocal();
    loaded=false;
    playBtn.textContent='\u25b6';
    playBtn.disabled=false;
  }
  if(!playBtn.dataset.cwStopBound){
    playBtn.dataset.cwStopBound='1';
    document.addEventListener('cosmos-world-music-stop',function(ev){handleExternalStop(ev.detail);});
    window.addEventListener('storage',function(ev){
      if(ev.key!=='cosmos-world-music-stop'||!ev.newValue)return;
      try{handleExternalStop(JSON.parse(ev.newValue));}catch(e){}
    });
    try{
      if(!window._cwBroadcast)window._cwBroadcast=new BroadcastChannel('cosmos-world-music-control');
      window._cwBroadcast.addEventListener('message',function(ev){handleExternalStop(ev.data);});
    }catch(e){}
  }

  function tryGD(kw){
    return fetch('https://music-api.gdstudio.xyz/api.php?types=search&name='+encodeURIComponent(kw)+'&count=1&pages=1')
      .then(function(r){return r.json();})
      .then(function(data){
        if(!data||!data[0]||!data[0].id)throw new Error('no result');
        return fetch('https://music-api.gdstudio.xyz/api.php?types=url&id='+data[0].id+'&br=128000');
      })
      .then(function(r){return r.json();})
      .then(function(d){
        var url=d.url||(d.data&&d.data.url);
        if(!url)throw new Error('no url');
        return url;
      });
  }

  function tryVKeys(kw){
    return fetch('https://api.vkeys.cn/v2/music/tencent?word='+encodeURIComponent(kw)+'&choose=1')
      .then(function(r){return r.json();})
      .then(function(d){
        var url=d.data&&d.data.url;
        if(!url)throw new Error('no url');
        return url;
      });
  }

  function loadAndPlay(){
    var myId=++cw.playId;
    announceStop();
    stopLocal();
    loaded=false;
    playBtn.textContent='\u2026';
    playBtn.disabled=true;
    cw.activeBtn=playBtn;

    cw.timer=setTimeout(function(){
      if(myId!==cw.playId)return;
      tryGD(keyword)
        .catch(function(){return tryVKeys(keyword);})
        .then(function(url){
          if(myId!==cw.playId)return;
          audio.src=url;
          loaded=true;
          cw.activeBtn=playBtn;
          playBtn.disabled=false;
          return audio.play();
        })
        .catch(function(){
          if(myId!==cw.playId)return;
          loaded=true;
          playBtn.disabled=false;
          playBtn.textContent='\u25b6';
        });
    },400);
  }

  if(!playBtn.dataset.cwClickBound){
    playBtn.dataset.cwClickBound='1';
    playBtn.addEventListener('click',function(ev){
      ev.preventDefault();ev.stopPropagation();
      if(playBtn.disabled)return;
      if(!loaded){loadAndPlay();return;}
      if(!audio.paused){
        audio.pause();
        playBtn.textContent='\u25b6';
      }else{
        announceStop();
        cw.activeBtn=playBtn;
        audio.play();
        playBtn.textContent='\u23f8';
      }
    });
  }

  setTimeout(function(){loadAndPlay();},50);
})();
</script>
</body>
</html>
