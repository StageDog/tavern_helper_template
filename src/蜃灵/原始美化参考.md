<script>
(function(){
  // 主题切换
  var KEY='cosmos-cw-theme';
  function safeGet(){try{return localStorage.getItem(KEY)||'dark';}catch(e){return 'dark';}}
  function safeSet(v){try{localStorage.setItem(KEY,v);}catch(e){}}
  function apply(t){document.querySelectorAll('.cosmos-world-bar').forEach(function(el){el.setAttribute('data-theme',t);});}
  apply(safeGet());
  document.querySelectorAll('.cosmos-world-bar').forEach(function(card){
    var btn=card.querySelector('.cosmos-cw-theme-btn');
    if(!btn)return;
    btn.addEventListener('click',function(ev){
      ev.preventDefault();ev.stopPropagation();
      var cur=card.getAttribute('data-theme')||safeGet()||'dark';
      var nxt=(cur==='dark')?'light':'dark';
      safeSet(nxt);apply(nxt);
    });
  });

  // 音乐播放器
  var lyricEl = document.querySelector('.cosmos-music-text');
  var playBtn = document.querySelector('.cosmos-music-btn');
  if(!lyricEl || !playBtn) return;

if(!window._cosmosAudio){
  window._cosmosAudio = new Audio();
  window._cosmosAudio.loop = true;
}
var audio = window._cosmosAudio;
  var loaded = false;
  var playing = false;

  var txt = lyricEl.textContent.trim();
  var m = txt.match(/[《<]([^》>]+)[》>]\s*(.+)/);
  if(!m){ playBtn.style.display='none'; return; }

  var keyword = m[1].trim() + ' ' + m[2].trim();

  // GD源（网易云）
  function tryGD(kw){
    return fetch('https://music-api.gdstudio.xyz/api.php?types=search&name='+encodeURIComponent(kw)+'&count=1&pages=1')
      .then(function(r){return r.json();})
      .then(function(data){
        if(!data||!data[0]||!data[0].id) throw new Error('no result');
        return fetch('https://music-api.gdstudio.xyz/api.php?types=url&id='+data[0].id+'&br=128000');
      })
      .then(function(r){return r.json();})
      .then(function(d){
        var url=d.url||(d.data&&d.data.url);
        if(!url) throw new Error('no url');
        return url;
      });
  }

  // VKeys源（QQ音乐）
  function tryVKeys(kw){
    return fetch('https://api.vkeys.cn/v2/music/tencent?word='+encodeURIComponent(kw)+'&choose=1')
      .then(function(r){return r.json();})
      .then(function(d){
        var url=d.data&&d.data.url;
        if(!url) throw new Error('no url');
        return url;
      });
  }

  function loadAndPlay(){
  audio.pause();
  audio.src='';
  loaded=false;
  playing=false;
    playBtn.textContent='…';
    playBtn.disabled=true;
    // 先试GD，失败fallback到VKeys
    tryGD(keyword)
      .catch(function(){return tryVKeys(keyword);})
      .then(function(url){
        audio.src=url;
        loaded=true;
        playBtn.disabled=false;
        return audio.play();
      })
      .then(function(){
        playing=true;
        playBtn.textContent='⏸';
      })
      .catch(function(){
        // 自动播放被拦截，等用户手动点
        loaded=true;
        playing=false;
        playBtn.disabled=false;
        playBtn.textContent='▶';
      });
  }

  playBtn.addEventListener('click',function(ev){
    ev.preventDefault();ev.stopPropagation();
    if(playBtn.disabled) return;
    if(!loaded){loadAndPlay();return;}
    if(playing){
      audio.pause();
      playing=false;
      playBtn.textContent='▶';
    } else {
      audio.play();
      playing=true;
      playBtn.textContent='⏸';
    }
  });

  // 自动尝试播放
  loadAndPlay();
})();
</script>
